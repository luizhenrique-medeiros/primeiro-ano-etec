import java.util.*;

public class Main
{
	public static void main(String[] args) {
		Scanner leia = new Scanner(System.in);
		int cont,soma=0,num;
		System.out.println("digite um número positivo");
		num=leia.nextInt();
		for(cont=1;cont<=num;cont++){
		    soma=soma+cont;
		    System.out.println(cont);
		}
		System.out.println("soma="+ soma);
	}
}
